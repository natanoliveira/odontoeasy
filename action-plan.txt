Comandos Úteis:

  # Ver dados no Prisma Studio
  npm run prisma:studio

  # Regenerar Prisma Client (após alterar schema)
  npx prisma generate --schema=src/prisma/schema.prisma

  # Reset banco (DEV ONLY!)
  npx prisma migrate reset --schema=src/prisma/schema.prisma

  # Nova migration
  npx prisma migrate dev --name nome_da_migration --schema=src/prisma/schema.prisma

  # Rodar seed manualmente
  npm run prisma:seed
